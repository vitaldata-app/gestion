ENRUTADOR GET (doGet) // ========================================== function doGet(e) { try { const params = e.parameter || {}; const accion = params.accion || params.action || "obtenerConfig"; const ssId = params.id; if (!ssId && accion !== "validarAcceso") { return responderJSON({ error: true, mensaje: "Falta el ID del consultorio." }); } if (accion === "validarAcceso") { return responderJSON(validarAccesoCliente(params.correo, params.clave)); } const ss = SpreadsheetApp.openById(ssId); switch (accion) { case "obtenerConfig": return responderJSON(obtenerConfiguracionClinica(ss, ssId)); case "buscarPaciente": return responderJSON(buscarPacientePorDni(ss, params.dni || params.numDoc)); case "buscarHistorial": return responderJSON(buscarHistorialClinico(ss, params.dni || params.numDoc)); case "obtenerSeguimiento": return responderJSON(obtenerListaSeguimiento(ss)); default: return responderJSON({ error: true, mensaje: "Acción no reconocida." }); } } catch (err) { return responderJSON({ error: true, mensaje: err.toString() }); } } // ========================================== // 2. ENRUTADOR POST (doPost) // ========================================== function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); const params = e.parameter || {}; const payload = JSON.parse(e.postData.contents || "{}"); const accion = params.accion || params.action || payload.action || payload.accion; const ssId = params.id || payload.spreadsheetId; if (!ssId) { return responderJSON({ error: true, mensaje: "Falta el ID de la base de datos." }); } const ss = SpreadsheetApp.openById(ssId); switch (accion) { case "registrarAdmision": return responderJSON(guardarAdmisionCompleta(ss, payload)); case "marcarContactado": return responderJSON(marcarSeguimientoContactado(ss, payload.idSeg)); case "obtenerMetas": return responderJSON(obtenerMetasFinancieras(ss, ssId, payload.pin)); case "actualizarTareas": return responderJSON(actualizarTareasMetas(ss, payload.tareas)); default: return responderJSON({ error: true, mensaje: "Acción POST no reconocida." }); } } catch (err) { return responderJSON({ error: true, mensaje: err.toString() }); } finally { lock.releaseLock(); } } // ========================================== // 3. MÓDULOS DE LECTURA Y CONFIGURACIÓN // ========================================== function obtenerConfiguracionClinica(ss, ssId) { const ssMaster = SpreadsheetApp.getActiveSpreadsheet(); const shClientes = ssMaster.getSheetByName("Clientes"); let clinicaNombre = ss.getName().replace("BD - ", ""); let clinicaLogo = ""; if (shClientes) { const dataCli = shClientes.getDataRange().getValues(); for (let i = 1; i < dataCli.length; i++) { if (String(dataCli[i][2]).trim() === String(ssId).trim()) { clinicaNombre = dataCli[i][1] || clinicaNombre; clinicaLogo = dataCli[i][4] || ""; break; } } } const shConfig = ss.getSheetByName("CONFIGURACIÓN"); const servicios = []; if (shConfig) { const dataCfg = shConfig.getDataRange().getValues(); for (let i = 1; i < dataCfg.length; i++) { if (dataCfg[i][0]) { servicios.push({ nombre: String(dataCfg[i][0]).trim(), categoria: String(dataCfg[i][1] || "General").trim(), pvp: Number(dataCfg[i][2]) || 0 }); } } } return { clinicaNombre, clinicaLogo, servicios }; } function buscarPacientePorDni(ss, numDoc) { const sh = ss.getSheetByName("PACIENTES"); if (!sh) return { encontrado: false }; const data = sh.getDataRange().getValues(); const busqueda = String(numDoc || "").trim(); for (let i = 1; i < data.length; i++) { if (String(data[i][1]).replace("'", "").trim() === busqueda) { const nomCompleto = String(data[i][2] || "").trim(); const partes = nomCompleto.split(" "); return { encontrado: true, data: { tipoDoc: data[i][0], numDoc: data[i][1], nombre: partes[0] || "", apePat: partes[1] || "", apeMat: partes.slice(2).join(" ") || "", celular: data[i][3], edad: data[i][4], distrito: "" } }; } } return { encontrado: false }; } function buscarHistorialClinico(ss, numDoc) { const sh = ss.getSheetByName("HISTORIAL_CLINICO"); if (!sh) return []; const data = sh.getDataRange().getValues(); const busqueda = String(numDoc || "").trim(); const historial = []; for (let i = data.length - 1; i >= 1; i--) { if (String(data[i][3]).replace("'", "").trim() === busqueda) { historial.push({ fecha: Utilities.formatDate(new Date(data[i][1]), "GMT-5", "dd/MM/yyyy"), servicio: data[i][5], producto: data[i][7] || data[i][5], diagnostico: data[i][6] }); if (historial.length >= 5) break; } } return historial; } function obtenerListaSeguimiento(ss) { const sh = ss.getSheetByName("SEGUIMIENTO"); if (!sh) return []; const data = sh.getDataRange().getValues(); const hoy = new Date(); hoy.setHours(0, 0, 0, 0); const res = []; for (let i = 1; i < data.length; i++) { const estado = String(data[i][7] || "").toUpperCase(); if (estado !== "CONTACTADO" && estado !== "ATENDIDO" && data[i][6]) { const fProg = new Date(data[i][6]); fProg.setHours(0, 0, 0, 0); const diffDias = Math.round((fProg - hoy) / (1000 * 60 * 60 * 24)); let categoria = "proximo"; if (diffDias < 0) categoria = "atrasado"; else if (diffDias === 0 || diffDias === 1) categoria = "hoy_manana"; res.push({ idSeg: data[i][0] || ("SEG-" + i), nombre: data[i][3], motivo: data[i][5], fechaStr: Utilities.formatDate(fProg, "GMT-5", "dd/MM/yyyy"), celular: data[i][4], categoria: categoria }); } } return res; } // ========================================== // 4. MÓDULOS TRANSACCIONALES DE ESCRITURA // ========================================== function guardarAdmisionCompleta(ss, p) { const ahora = new Date(); const fechaStr = Utilities.formatDate(ahora, "GMT-5", "dd/MM/yyyy"); const horaStr = Utilities.formatDate(ahora, "GMT-5", "HH:mm:ss"); const idTx = "TX-" + Utilities.getUuid().substring(0, 8).toUpperCase(); const nombreCompleto = `${p.nombre} ${p.apellidoP} ${p.apellidoM}`.trim(); // 1. Pacientes const shPac = ss.getSheetByName("PACIENTES"); if (shPac) { const dataPac = shPac.getDataRange().getValues(); let existe = false; for (let i = 1; i < dataPac.length; i++) { if (String(dataPac[i][1]).replace("'", "").trim() === String(p.numDoc).trim()) { existe = true; break; } } if (!existe) { shPac.appendRow([p.tipoDoc || "DNI", "'" + p.numDoc, nombreCompleto, p.celular || "", p.edad || "", fechaStr]); } } // 2. Data Madre y Seguimiento const shData = ss.getSheetByName("DATA_MADRE"); const shSeg = ss.getSheetByName("SEGUIMIENTO"); const descPorItem = (Number(p.descuento) || 0) / (p.carrito.length || 1); p.carrito.forEach(item => { const pvp = Number(item.pvp || item.precio || 0); const neto = Math.max(0, pvp - descPorItem); if (shData) { shData.appendRow([ fechaStr, p.tipoDoc || "DNI", "'" + p.numDoc, nombreCompleto, item.categoria || "Servicio", item.nombre, pvp, 1, descPorItem, neto, p.tipoPago || "Efectivo", "PAGADO", "Obstetra Turno", idTx, p.origen || "Directo", horaStr ]); } if (item.fechaSeguimiento && shSeg) { shSeg.appendRow([ "SEG-" + Utilities.getUuid().substring(0, 6).toUpperCase(), fechaStr, "'" + p.numDoc, nombreCompleto, p.celular || "", "Control " + item.nombre, item.fechaSeguimiento, "PENDIENTE" ]); } }); return { resultado: "OK", transaccionId: idTx }; } function marcarSeguimientoContactado(ss, idSeg) { const sh = ss.getSheetByName("SEGUIMIENTO"); if (!sh) return { resultado: "OK" }; const data = sh.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === String(idSeg).trim()) { sh.getRange(i + 1, 8).setValue("CONTACTADO"); break; } } return { resultado: "OK" }; } function obtenerMetasFinancieras(ss, ssId, pinIngresado) { const ssMaster = SpreadsheetApp.getActiveSpreadsheet(); const shClientes = ssMaster.getSheetByName("Clientes"); let pinValido = "1234"; if (shClientes) { const dataCli = shClientes.getDataRange().getValues(); for (let i = 1; i < dataCli.length; i++) { if (String(dataCli[i][2]).trim() === String(ssId).trim()) { pinValido = String(dataCli[i][3]).trim(); break; } } } if (String(pinIngresado).trim() !== pinValido) { return { error: true, mensaje: "PIN incorrecto" }; } // Meta y Logrado const shMetas = ss.getSheetByName("METAS"); const metaMonto = Number(shMetas ? shMetas.getRange("B2").getValue() : 0) || 3000; const shData = ss.getSheetByName("DATA_MADRE"); let recaudadoSemanal = 0; if (shData) { const ventas = shData.getDataRange().getValues(); const hoy = new Date(); const lunes = new Date(hoy.setDate(hoy.getDate() - (hoy.getDay() === 0 ? 6 : hoy.getDay() - 1))); lunes.setHours(0, 0, 0, 0); for (let i = 1; i < ventas.length; i++) { if (ventas[i][0]) { const parts = String(ventas[i][0]).split("/"); const fVenta = parts.length === 3 ? new Date(parts[2], parts[1] - 1, parts[0]) : new Date(ventas[i][0]); if (fVenta >= lunes) recaudadoSemanal += Number(ventas[i][9]) || 0; } } } const tareas = []; if (shMetas) { const dataM = shMetas.getDataRange().getValues(); for (let j = 1; j < dataM.length; j++) { if (dataM[j][4]) { tareas.push({ id: dataM[j][2] || ("TSK-" + j), descripcion: dataM[j][4], completado: String(dataM[j][5]).toUpperCase() === "COMPLETADO" }); } } } const pct = metaMonto > 0 ? Math.round((recaudadoSemanal / metaMonto) * 100) : 0; return { metaFinanciera: { meta: metaMonto, logrado: recaudadoSemanal, porcentaje: pct }, objetivosTareas: tareas }; } function actualizarTareasMetas(ss, tareas) { const sh = ss.getSheetByName("METAS"); if (!sh || !tareas) return { resultado: "OK" }; const data = sh.getDataRange().getValues(); tareas.forEach(t => { for (let i = 1; i < data.length; i++) { if (String(data[i][4]).trim() === String(t.descripcion).trim()) { sh.getRange(i + 1, 6).setValue(t.completado ? "COMPLETADO" : "PENDIENTE"); break; } } }); return { resultado: "OK" }; } // ========================================== // 5. PIPELINE ETL CONSOLIDADOR (OLAP / POWER BI) // ========================================== function ejecutarPipelineETL() { const ssMaster = SpreadsheetApp.getActiveSpreadsheet(); const shClientes = ssMaster.getSheetByName("Clientes"); let shConsolidada = ssMaster.getSheetByName("DM_DATA_CONSOLIDADA"); if (!shConsolidada) shConsolidada = ssMaster.insertSheet("DM_DATA_CONSOLIDADA"); const clientes = shClientes.getDataRange().getValues(); const registrosConsolidados = []; const cabeceras = [ "ID_CLIENTE_CORREO", "NOMBRE_EMPRESA", "ID_TRANSACCION", "FECHA", "HORA", "MES", "ANO", "TIPO_DOC", "NUM_DOC", "PACIENTE", "CATEGORIA", "ITEM_SERVICIO", "PRECIO_UNITARIO", "CANTIDAD", "DESCUENTO", "VENTA_NETA", "MEDIO_PAGO", "ESTADO_PAGO", "ATENDIDO_POR", "ORIGEN" ]; if (estado === "ACTIVO" && ssIdCliente) { try { const ssCli = SpreadsheetApp.openById(ssIdCliente); const shData = ssCli.getSheetByName("DATA_MADRE"); if (shData) { const rows = shData.getDataRange().getValues(); registrosConsolidados.push([ correo, empresa, rows[r][13], rows[r][0], rows[r][15], mesStr, anoStr, rows[r][1], rows[r][2], rows[r][3], rows[r][4], rows[r][5], rows[r][6], rows[r][7], rows[r][8], rows[r][9], rows[r][10], rows[r][11], rows[r][12], rows[r][14] ]); shConsolidada.getRange(1, 1, 1, cabeceras.length).setBackground("#0f172a").setFontColor("#ffffff").setFontWeight("bold"); } } function responderJSON(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
